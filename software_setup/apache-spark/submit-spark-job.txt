from sedona.spark.SedonaContext import SedonaContext
sedona = SedonaContext.create(spark)

spark.sql("""
CREATE TABLE IF NOT EXISTS local.flood_db.tile_metadata (
    tile_id STRING,       -- Unique ID for the SAR tile
    tile_path STRING,     -- Where the Parquet file lives
    bbox BINARY,          -- The spatial bounding box (math)
    processed_at TIMESTAMP -- When the mask was created
) 
USING iceberg
""")

# 1. SAR Metadata: Orbit path, Sensor (Sentinel-1), and the Footprint
# These coordinates represent a typical 20km x 20km SAR tile footprint
lon_min, lat_min = 76.50, 12.10
lon_max, lat_max = 76.70, 12.30

spark.sql(f"""
INSERT INTO local.flood_db.tile_metadata
VALUES (
    'S1A_IW_GRDH_20240124_MYS',         -- Standard SAR Product ID
    '/data/masks/mysore_flood_v1.parquet', -- Path to processed binary mask
    ST_AsBinary(ST_MakeEnvelope({lon_min}, {lat_min}, {lon_max}, {lat_max})), 
    current_timestamp()
)
""")


# Query: Find SAR tiles where the footprint 'Contains' our point of interest
target_lon = 76.55
target_lat = 12.15

spark.sql(f"""
SELECT 
    tile_id, 
    processed_at
FROM local.flood_db.tile_metadata
WHERE ST_Contains(ST_GeomFromWKB(bbox), ST_Point({target_lon}, {target_lat}))
""").show()
